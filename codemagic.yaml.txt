workflows:
  android-debug-apk:
    name: Android Debug APK
    max_build_duration: 30
    instance_type: linux_x2
    environment:
      groups: []
      vars:
        JAVA_HOME: /usr/lib/jvm/jdk-17.0.x
      java: 17
    triggering:
      events:
        - push
      branch_patterns:
        - pattern: main
          include: true
          source: true
    scripts:
      - name: Generate Gradle wrapper (if missing)
        script: |
          if [ ! -f "./gradlew" ]; then
            gradle wrapper --gradle-version 8.7
          fi
      - name: Set execute permission for gradlew
        script: chmod +x ./gradlew
      - name: Build debug APK
        script: ./gradlew assembleDebug --stacktrace
    artifacts:
      - app/build/outputs/apk/**/*.apk
      - app/build/outputs/**/mapping.txt
    publishing:
      email:
        recipients:
          - your-email@example.com
        notify:
          success: true
          failure: true
